-- ===================== DRAG =====================
do
    local function registerDrag(frame, dragHandle)
        local dragging, dragStart, startPos = false, Vector2.new(), Vector2.new()
        dragHandle.InputBegan:Connect(function(input)
            if input.UserInputType==Enum.UserInputType.MouseButton1 or input.UserInputType==Enum.UserInputType.Touch then
                dragging=true
                dragStart=Vector2.new(input.Position.X, input.Position.Y)
                startPos=Vector2.new(frame.Position.X.Offset, frame.Position.Y.Offset)
            end
        end)
        UIS.InputEnded:Connect(function(input)
            if input.UserInputType==Enum.UserInputType.MouseButton1 or input.UserInputType==Enum.UserInputType.Touch then
                dragging=false
            end
        end)
        UIS.InputChanged:Connect(function(input)
            if dragging and (input.UserInputType==Enum.UserInputType.MouseMovement or input.UserInputType==Enum.UserInputType.Touch) then
                local delta = Vector2.new(input.Position.X, input.Position.Y) - dragStart
                frame.Position = UDim2.new(frame.Position.X.Scale, math.round(startPos.X + delta.X), frame.Position.Y.Scale, math.round(startPos.Y + delta.Y))
            end
        end)
    end
    getgenv().SW_registerDrag = registerDrag
end

    local function AddToggle(parent, text, flag, default, defaultKey, callback)
        local Flags = getgenv().SW_Flags
        local Controls = getgenv().SW_Controls
        local ToggleCallbacks = getgenv().SW_ToggleCallbacks
        local KeybindRegistry = getgenv().SW_KeybindRegistry
        local FlagKeys = getgenv().SW_FlagKeys
        local RefreshKeybindList = getgenv().SW_RefreshKeybindList
        Flags[flag]=default or false
        local tb=Instance.new("TextButton"); tb.Size=UDim2.new(1,0,0,20); tb.BackgroundTransparency=1; tb.Text=""; tb.Parent=parent
        local lbl=Instance.new("TextLabel"); lbl.Size=UDim2.new(1,-70,1,0); lbl.BackgroundTransparency=1
        lbl.Font=Enum.Font.GothamSemibold; lbl.TextSize=11; lbl.TextColor3=Theme.TextMain
        lbl.TextXAlignment=Enum.TextXAlignment.Left; lbl.Text=text; lbl.Parent=tb
        local kb=Instance.new("TextButton"); kb.Size=UDim2.new(0,32,0,14); kb.Position=UDim2.new(1,-66,0.5,-7)
        kb.BackgroundColor3=Theme.ToggleOff; kb.BorderSizePixel=0; kb.Font=Enum.Font.GothamBold
        kb.TextSize=9; kb.TextColor3=Theme.TextSub; kb.Text=defaultKey or "NONE"; kb.Parent=tb
    if hasTouch then kb.Visible = false end
        local kbc=Instance.new("UICorner"); kbc.CornerRadius=UDim.new(0,3); kbc.Parent=kb
        local track=Instance.new("Frame"); track.Size=UDim2.new(0,28,0,14); track.Position=UDim2.new(1,-28,0.5,-7)
        track.BackgroundColor3=Flags[flag] and Theme.AccentWhite or Theme.ToggleOff; track.BorderSizePixel=0; track.Parent=tb
        local tc=Instance.new("UICorner"); tc.CornerRadius=UDim.new(1,0); tc.Parent=track
        local knob=Instance.new("Frame"); knob.Size=UDim2.new(0,10,0,10)
        knob.Position=Flags[flag] and UDim2.new(1,-12,0.5,-5) or UDim2.new(0,2,0.5,-5)
        knob.BackgroundColor3=Flags[flag] and Color3.fromRGB(20,20,20) or Color3.fromRGB(200,200,200); knob.BorderSizePixel=0; knob.Parent=track
        local knc=Instance.new("UICorner"); knc.CornerRadius=UDim.new(1,0); knc.Parent=knob
        local state=Flags[flag]; local curKey=defaultKey

        local function applyBind(newKey)
            curKey = newKey; FlagKeys[flag] = newKey
            kb.Text = (newKey and newKey ~= "NONE") and newKey or "NONE"
            if newKey and newKey ~= "NONE" then KeybindRegistry[flag] = {Key=newKey, Active=state}
            else KeybindRegistry[flag] = nil end
            RefreshKeybindList()
        end

        local function setState(ns)
            state=ns; Flags[flag]=state
            Tween(knob,TweenInfo.new(0.2),{Position=state and UDim2.new(1,-12,0.5,-5) or UDim2.new(0,2,0.5,-5),
                BackgroundColor3=state and Color3.fromRGB(20,20,20) or Color3.fromRGB(200,200,200)})
            Tween(track,TweenInfo.new(0.2),{BackgroundColor3=state and Theme.AccentWhite or Theme.ToggleOff})
            if curKey and curKey~="NONE" then KeybindRegistry[flag]={Key=curKey,Active=state}; RefreshKeybindList() end
            if callback then callback(state) end
        end
        ToggleCallbacks[flag]=setState
        if curKey and curKey~="NONE" then FlagKeys[flag]=curKey; KeybindRegistry[flag]={Key=curKey,Active=state}; RefreshKeybindList() end
        tb.MouseButton1Click:Connect(function() setState(not state) end)

        local listening=false
        kb.MouseButton1Down:Connect(function()
            if listening then listening=false; applyBind("NONE"); return end
            listening=true; kb.Text="..."
        end)
        kb.MouseButton2Click:Connect(function() listening=false; applyBind("NONE") end)
        UIS.InputBegan:Connect(function(input, gp)
            if not listening then return end
            if input.UserInputType==Enum.UserInputType.Keyboard then
                listening=false; applyBind(input.KeyCode.Name)
            elseif input.UserInputType==Enum.UserInputType.MouseButton1 or input.UserInputType==Enum.UserInputType.Touch then
                local CG = getgenv().SW_CanvasGroupRef
                if CG and CG.Visible then
                    local ap=CG.AbsolutePosition; local as=CG.AbsoluteSize
                    local px,py=input.Position.X, input.Position.Y
                    local inside = px>=ap.X and px<=ap.X+as.X and py>=ap.Y and py<=ap.Y+as.Y
                    if not inside then listening=false; applyBind("NONE") end
                end
            end
        end)
        Controls[flag]={Set=setState, SetKey=applyBind, GetKey=function() return curKey end}
    end

        for _, item in pairs(ItemData.Items) do
            if type(item) == "table" and type(item.ArrowConfig) == "table" then
                local ac = item.ArrowConfig
                if noDrop then
                    ac.Gravity = 0; ac.MaxTime = 999
                else
                    if ac.__origGravity ~= nil then ac.Gravity = ac.__origGravity end
                    if ac.__origMaxTime ~= nil then ac.MaxTime = ac.__origMaxTime end
                end
                if noDrop or noSpread then
                    ac.Spread = 0
                else
                    if ac.__origSpread ~= nil then ac.Spread = ac.__origSpread end
                end
            end
        end
    end
    getgenv().SW_applyArrowOnce = applyArrowOnce
    local function ensureArrowLoop()
        local needLoop = getgenv().SW_Flags["combat_noblulletdrop"] or getgenv().SW_Flags["combat_nospread"]
        if needLoop and not arrowLoopTask then
            arrowLoopTask = task.spawn(function()
                while SW.Alive do
                    if not (getgenv().SW_Flags["combat_noblulletdrop"] or getgenv().SW_Flags["combat_nospread"]) then
                        arrowLoopTask = nil; return
                    end
                    applyArrowOnce()
                    task.wait(0.15)
                end
            end)
        elseif not needLoop then
            applyArrowOnce(); arrowLoopTask = nil
        end
    end
    getgenv().SW_ensureArrowLoop = ensureArrowLoop
end)

    local originalBowItemData = {}
    local function applyBowTimings()
        local enabled = getgenv().SW_InstantBowEnabled
        pcall(function()
            local ItemDataModule = ReplicatedStorage:FindFirstChild("ItemData")
            if not ItemDataModule then return end
            local ItemData = require(ItemDataModule)
            for name, item in pairs(ItemData.Items) do
                if type(item) == "table" and (name:lower():find("bow") or name:lower():find("cross")) then
                    if not originalBowItemData[name] then
                        originalBowItemData[name] = {
                            ReloadInfo = type(item.ReloadInfo) == "table" and table.clone(item.ReloadInfo) or nil,
                            ToolInfo   = type(item.ToolInfo) == "table" and table.clone(item.ToolInfo) or nil,
                            ArrowConfig= type(item.ArrowConfig) == "table" and table.clone(item.ArrowConfig) or nil,
                        }
                    end
                    local orig = originalBowItemData[name]
                    if type(item.ReloadInfo) == "table" and orig.ReloadInfo then
                        item.ReloadInfo.ReloadDuration  = enabled and 0.01 or orig.ReloadInfo.ReloadDuration
                        item.ReloadInfo.ReturnsMagTime  = enabled and 0.01 or orig.ReloadInfo.ReturnsMagTime
                        item.ReloadInfo.GetsMagTime     = enabled and 0.01 or orig.ReloadInfo.GetsMagTime
                    end
                    if type(item.ToolInfo) == "table" and orig.ToolInfo then
                        item.ToolInfo.HitTime  = enabled and 0.01 or orig.ToolInfo.HitTime
                        item.ToolInfo.PlayTime = enabled and 0.01 or orig.ToolInfo.PlayTime
                        item.ToolInfo.WaitTime = enabled and 0.01 or orig.ToolInfo.WaitTime
                        if item.ToolInfo.ChargeTime ~= nil then
                            item.ToolInfo.ChargeTime = enabled and 0.01 or orig.ToolInfo.ChargeTime
                        end
                    end
                    if type(item.ArrowConfig) == "table" and orig.ArrowConfig then
                        item.ArrowConfig.StraightTime = enabled and 0     or orig.ArrowConfig.StraightTime
                        item.ArrowConfig.MaxTime      = enabled and 999   or orig.ArrowConfig.MaxTime
                        item.ArrowConfig.Gravity      = enabled and 0     or orig.ArrowConfig.Gravity
                        item.ArrowConfig.Spread       = enabled and 0     or orig.ArrowConfig.Spread
                    end
                end
            end
        end)
    end
    getgenv().SW_applyBowTimings = applyBowTimings

    local function getTool()
        local ch = LocalPlayer.Character
        return ch and ch:FindFirstChildOfClass("Tool")
    end

        if not tool then return end

        if rts then rts.Value = true end
    end

                cfg.Gravity = 0; cfg.Spread = 0; cfg.StraightTime = 0; cfg.MaxTime = 999; cfg.MaxDistance = 99999
            end
            return cfg
        end

        local oldAim = BowShootModule.aim
        BowShootModule.aim = function(isAiming, viewmodel, tool, chargeValue, holdAnim, itemHoldAnim, readyAnim, isBow)
            if not getgenv().SW_InstantBowEnabled then
                return oldAim(isAiming, viewmodel, tool, chargeValue, holdAnim, itemHoldAnim, readyAnim, isBow)
            end
            if not isAiming then
                pcall(oldAim, isAiming, viewmodel, tool, chargeValue, holdAnim, itemHoldAnim, readyAnim, isBow)
                hideArrowParts(tool)
                return
            end
            if viewmodel then showArrowParts(viewmodel) end
            if tool then showArrowParts(tool) end
            task.spawn(function()
                pcall(oldAim, isAiming, viewmodel, tool, chargeValue, holdAnim, itemHoldAnim, readyAnim, isBow)
            end)
            forceReady(tool)
        end

        local oldShoot = BowShootModule.Shoot
        local lastAmmoCheck = {}
        BowShootModule.Shoot = function(slot, toolData)
            if not getgenv().SW_InstantBowEnabled then return oldShoot(slot, toolData) end
            local toolModel = toolData and toolData.ToolModel
            if toolModel then
                local now = tick()
                local last = lastAmmoCheck[toolModel] or 0
                if now - last > 0.5 then
                    lastAmmoCheck[toolModel] = now
                    for _, v in ipairs(toolModel:GetDescendants()) do
                        if v:IsA("IntValue") or v:IsA("NumberValue") then
                            local n = v.Name:lower()
                            if (n:find("ammo") or n:find("mag") or n:find("arrow") or n:find("loaded")) and v.Value <= 0 then
                                return false
                            end
                        end
                    end
                end
            end
            forceReady(toolModel)
            local ok, res = pcall(oldShoot, slot, toolData)
            task.spawn(function() forceReady(toolModel) end)
            return ok and res or false
        end
    end)

            end

                    local args={...}; local data=args[1]
                    if name=="BowShoot" and Flags["combat_instantbow"] then
                        for i,a in ipairs(args) do
                            if type(a)=="number" and a>=0 and a<=1.01 then args[i]=1
                            elseif type(a)=="table" then
                                for k,v in pairs(a) do
                                    if type(v)=="number" and v>=0 and v<=1.01 then a[k]=1 end
                                end
                            end
                        end
                    end
                    if type(data)=="table" and data.CameraOrigin and data.CameraDirection then
                        table.insert(getgenv().SW_pendingTracers,{origin=data.CameraOrigin, dir=data.CameraDirection.Unit})
                    end
                    if Flags["vis_tb"] and name=="Shot" and type(data)=="table" and data.MuzzleOrigin and data.CameraOrigin and data.CameraDirection then

                            pcall(function()

                                tbRaycast.IgnoreWater = true
                                local ignore = {Camera}
                                if LocalPlayer.Character then table.insert(ignore, LocalPlayer.Character) end
                                tbRaycast.FilterDescendantsInstances = ignore
                                local res = workspace:Raycast(data.CameraOrigin, data.CameraDirection * 1500, tbRaycast)
                                local hit = res and res.Position or (data.CameraOrigin + data.CameraDirection * 1500)

        local crosshairExtra = Flags["vis_crosshair"] and (crosshairRadius + 6) or 0
        local baseOffset = Flags["vis_targethud_offset"] or 16
        TargetHolder.Position = UDim2.fromOffset(center.X, center.Y + baseOffset + crosshairExtra)
        local fov = Flags["vis_targethud_fov"] or 160
        local maxDist = Flags["vis_targethud_dist"] or 500
        local bestPlayer, minDistance = nil, fov
        local myChar = LocalPlayer.Character
        local myRoot = myChar and (myChar:FindFirstChild("HumanoidRootPart") or myChar:FindFirstChild("Head"))
        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= LocalPlayer and p.Character then
                local targetPart = p.Character:FindFirstChild("Head") or p.Character:FindFirstChild("HumanoidRootPart") or p.Character.PrimaryPart
                if targetPart then
                    local hum = p.Character:FindFirstChild("Humanoid")
                    local isAlive = not hum or (hum.Health > 0)
                    if isAlive then
                        local screenPos, onScreen = Camera:WorldToViewportPoint(targetPart.Position)
                        if onScreen and screenPos.Z > 0 then
                            local screenDist = (Vector2.new(screenPos.X, screenPos.Y) - center).Magnitude
                            if screenDist < minDistance then
                                if myRoot then
                                    local realDist = (myRoot.Position - targetPart.Position).Magnitude
                                    if realDist <= maxDist then minDistance = screenDist; bestPlayer = p end
                                else minDistance = screenDist; bestPlayer = p end
                            end
                        end
                    end
                end
            end
        end
        if bestPlayer then
            if cachedTarget ~= bestPlayer then cachedTarget = bestPlayer; UpdateTarget(bestPlayer) end
            TargetHolder.Visible = true
        else
            TargetHolder.Visible = false; cachedTarget = nil; cachedUserId = nil
        end
    end

        getgenv().SW_targetHudUpdate = nil
        getgenv().Fanity_CrosshairTarget_Cleanup = nil
    end

    local function ConvertScreenPoint(worldPos)
        local ViewportSize = Camera.ViewportSize
        local LocalPos = Camera.CFrame:pointToObjectSpace(worldPos)
        local AspectRatio = ViewportSize.X / ViewportSize.Y
        local HalfY = -LocalPos.Z * math.tan(math.rad(Camera.FieldOfView / 2))
        local HalfX = AspectRatio * HalfY
        local FarPlaneCorner = Vector3.new(-HalfX, HalfY, LocalPos.Z)
        local RelativePos = LocalPos - FarPlaneCorner
        local ScreenX = RelativePos.X / (HalfX * 2)
        local ScreenY = -RelativePos.Y / (HalfY * 2)
        local InFront = -LocalPos.Z > 0
        return Vector3.new(ScreenX * ViewportSize.X, ScreenY * ViewportSize.Y, -LocalPos.Z), InFront
    end
    local function BoxSolve(torso)
        if not torso then return nil end
        local ViewportTop = torso.Position + (torso.CFrame.UpVector * 1.8) + Camera.CFrame.UpVector
        local ViewportBottom = torso.Position - (torso.CFrame.UpVector * 2.5) - Camera.CFrame.UpVector
        local Distance = (torso.Position - Camera.CFrame.p).Magnitude
        local NewDistance = math.floor(Distance * 0.333)
        local Top, TopRendered = ConvertScreenPoint(ViewportTop)
        local Bottom, _ = ConvertScreenPoint(ViewportBottom)
        local Width = math.max(math.floor(math.abs(Top.X - Bottom.X)), 8)
        local Height = math.max(math.floor(math.max(math.abs(Bottom.Y - Top.Y), Width/2)), 12)
        local BoxSize = Vector2.new(math.floor(math.max(Height/1.5, Width)), Height)
        local BoxPosition = Vector2.new(math.floor(Top.X*0.5 + Bottom.X*0.5 - BoxSize.X*0.5), math.floor(math.min(Top.Y, Bottom.Y)))
        return BoxSize, BoxPosition, TopRendered, NewDistance
    end

    local function CheckPlayerHeldItem(p)
        local char = p.Character
        if not char then return "None" end
        local tool = char:FindFirstChildOfClass("Tool")
        if tool then
            local tName = tool:GetAttribute("ItemName") or tool:FindFirstChild("ItemName")
            if tName and tostring(tName.Value or tName) ~= "" then return tostring(tName.Value or tName) end
            local low = string.lower(tool.Name)
            if not HELD_BLACKLIST[low] and tool.Name ~= "Tool" then return tool.Name end
        end
        local data = p:FindFirstChild("Data") or char:FindFirstChild("Data")
        if data then
            local act = data:FindFirstChild("ActiveSlot") or data:FindFirstChild("EquippedSlot")
            if act then
                local slot = data:FindFirstChild("Slot" .. tostring(act.Value)) or data:FindFirstChild(tostring(act.Value))
                if slot then
                    local item = slot:FindFirstChild("ItemName") or slot:FindFirstChild("Item") or (slot:IsA("StringValue") and slot)
                    if item and item.Value ~= "" and item.Value ~= "None" then return item.Value end
                end
            end
        end
        local rHand = char:FindFirstChild("RightHand") or char:FindFirstChild("Right Arm")
        if rHand then
            for _, obj in ipairs(rHand:GetChildren()) do
                if obj:IsA("Weld") or obj:IsA("Motor6D") then
                    local other = (obj.Part0 == rHand and obj.Part1) or (obj.Part1 == rHand and obj.Part0)
                    if other and other.Parent ~= char and not other.Parent:IsA("Accessory") then
                        local name = other.Parent.Name
                        local low = string.lower(name)
                        if not low:find("armor") and not low:find("vest") and not low:find("kilt") and not HELD_BLACKLIST[low] then return name end
                    end
                end
            end
        end
        return "None"
    end

                    end
                end

        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= LocalPlayer and p.Character then
                local root = p.Character:FindFirstChild("HumanoidRootPart")
                if root then
                    local sPos, onScreen = Camera:WorldToViewportPoint(root.Position)
                    if onScreen and sPos.Z > 0 then
                        local dist = (Vector2.new(sPos.X, sPos.Y) - center).Magnitude
                        if dist < bestDist then bestDist = dist; bestP = p end

                end
            end

    local function GetStorageContents(storage)
        if not storage then return nil, 0 end
        local items, hasItems = {}, false
        for _, slot in pairs(storage:GetChildren()) do
            local itemName = slot:GetAttribute("Item") or slot:GetAttribute("ItemName")
            if not itemName then
                local str = slot:FindFirstChildWhichIsA("StringValue", true)
                if str then itemName = str.Value end
            end
            if itemName and itemName ~= "" and itemName ~= "Empty" then
                local count = slot:GetAttribute("Amount") or slot:GetAttribute("Count")
                if not count then
                    local num = slot:FindFirstChildWhichIsA("IntValue", true) or slot:FindFirstChildWhichIsA("NumberValue", true)
                    if num then count = num.Value end
                end
                count = tonumber(count) or 1
                items[itemName] = (items[itemName] or 0) + count
                hasItems = true
            end
        end
        return items, hasItems
    end

        local function processESPList(list, flagName, maxDist, filterName)
            local isEnabled = Flags[flagName]
            for obj, data in pairs(list) do
                if not obj.Parent then cleanWorldESP(data); list[obj] = nil; continue end
                local filterAllowed = true
                if filterName and Flags[filterName] then
                    if data.oType and not Flags[filterName][data.oType] then filterAllowed = false end
                    if data.cType and not Flags[filterName][data.cType] then filterAllowed = false end
                end
                if not isEnabled or not root or not filterAllowed then
                    if data.bb and data.bb.Enabled then data.bb.Enabled = false end
                    if data.hl and data.hl.Enabled then data.hl.Enabled = false end
                    continue
                end
                local dist = (rootPos - data.part.Position).Magnitude
                if dist <= maxDist then
                    data.bb.Enabled = true
                    if data.hl then data.hl.Enabled = true end
                    local baseText = data.oType or data.cType or data.txt.Text:split("\n")[1]
                    if flagName == "vis_cratestorage" then
                        if data.dirty or tickTime - data.lastUpdate > 5 then
                            data.lastUpdate = tickTime

                                data.txt.Text = table.concat(lines, "\n")

                                data.txt.Text = "[ Wooden Crate ] - " .. math.floor(dist) .. "s\nEmpty"
                                data.bb.Size = UDim2.fromOffset(200, 40)
                            end
                            data.dirty = false
                        end
                    else
                        data.txt.Text = string.format("%s\n%ds", baseText, math.floor(dist))
                    end
                else
                    data.bb.Enabled = false
                    if data.hl then data.hl.Enabled = false end
                end
            end
        end

        processESPList(TrackedNPCs, "vis_npc", 2000)
        processESPList(TrackedBoats, "vis_boats", 3000)
        processESPList(TrackedOres, "vis_ores", 2000, "vis_ores_filter")
        processESPList(TrackedCrates, "vis_crates", 3000, "vis_crates_filter")
        processESPList(TrackedStorages, "vis_cratestorage", 1500)
    end

    local function EnableSmoothDrag(frame, dragHandle)
        local dragging = false
        local dragStart = Vector2.new()
        local startPos = Vector2.new()
        local targetPos = Vector2.new(frame.Position.X.Offset, frame.Position.Y.Offset)

                if not frame.Visible then return end
                local currentPos = Vector2.new(frame.Position.X.Offset, frame.Position.Y.Offset)
                local lerped = currentPos:Lerp(targetPos, math.clamp(dt * 24, 0, 1))
                frame.Position = UDim2.new(frame.Position.X.Scale, math.round(lerped.X), frame.Position.Y.Scale, math.round(lerped.Y))
                if not dragging and (lerped - targetPos).Magnitude < 0.5 then
                    pcall(function() activeConn:Disconnect() end); activeConn = nil
                end
            end)
        end
        dragHandle.InputBegan:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                dragging = true; dragStart = Vector2.new(input.Position.X, input.Position.Y)
                startPos = Vector2.new(frame.Position.X.Offset, frame.Position.Y.Offset)
                startLoop()
            end
        end)
        UIS.InputEnded:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then dragging = false end
        end)
        UIS.InputChanged:Connect(function(input)
            if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
                local delta = Vector2.new(input.Position.X, input.Position.Y) - dragStart
                targetPos = startPos + delta
            end
        end)
    end

    local SKIP = { [""]=true, [" "]=true, Empty=true, ["0"]=true, None=true }
    local cachedItems = {}
    local hoverLastPlayer = nil
    local function AreIdentical(t1, t2)
        if not t1 or not t2 then return false end
        for k, v in pairs(t1) do if t2[k] ~= v then return false end end
        for k, v in pairs(t2) do if t1[k] ~= v then return false end end
        return true
    end
    local function UpdateInventoryUI(player, force)
        local dataFolder = player:FindFirstChild("Data")
        if not dataFolder then return end
        local currentItems = {}
        for _, slot in ipairs(dataFolder:GetChildren()) do
            local nameObj = slot:FindFirstChild("ItemName") or slot:FindFirstChild("Item") or slot:FindFirstChild("Name")
            local name
            if nameObj and nameObj:IsA("StringValue") then name = nameObj.Value
            elseif slot:IsA("StringValue") then name = slot.Value end
            if name then name = name:match("^%s*(.-)%s*$") end
            if name and not SKIP[name] and #name > 0 then
                local add = 1
                local amt = slot:FindFirstChild("Amount") or slot:FindFirstChild("Count") or slot:FindFirstChild("Quantity") or slot:FindFirstChild("Stack")
                if amt and (amt:IsA("IntValue") or amt:IsA("NumberValue")) and amt.Value > 0 then add = amt.Value end
                currentItems[name] = (currentItems[name] or 0) + add
            end
        end
        if not force and AreIdentical(cachedItems, currentItems) then return end
        cachedItems = currentItems
        for _, child in pairs(ItemScroll:GetChildren()) do
            if child:IsA("Frame") then child:Destroy() end
        end
        local totalItems = 0
        for itemName, count in pairs(currentItems) do
            CreateItemSlot(itemName, count)
            totalItems = totalItems + 1
        end
        SubText.Text = string.format("ITEMS COUNT: %d", totalItems)
        ItemScroll.CanvasSize = UDim2.new(0, 0, 0, math.ceil(math.max(1, totalItems) / 3) * 100)
    end
    local function SetTargetPlayer(player)
        TitleText.Text = string.format("<b>%s</b> <font color=\"#8A8F9E\">[Fanity.gg]</font>", player.DisplayName or player.Name)
        task.spawn(function()
            local thumb, isReady = Players:GetUserThumbnailAsync(player.UserId, Enum.ThumbnailType.HeadShot, Enum.ThumbnailSize.Size100x100)
            if thumb and Avatar then Avatar.Image = thumb end
        end)
    end

    getgenv().SW_hoverInvUpdate = function(dt)
        if not Flags["vis_hoverinv"] then
            if hoverLastPlayer then SetUIVisibility(false); hoverLastPlayer = nil; cachedItems = {} end
            return
        end
        hoverAcc = hoverAcc + (dt or 0.016)
        if hoverAcc < 0.1 then return end
        hoverAcc = 0
        local center = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
        local maxDist = Flags["vis_hoverinv_dist"] or 150
        local effectiveMaxDist = hoverLastPlayer and (maxDist + 40) or maxDist
        local closest, minDist = nil, effectiveMaxDist
        for _, p in pairs(Players:GetPlayers()) do
            if p ~= LocalPlayer and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
                local hum = p.Character:FindFirstChild("Humanoid")
                if hum and hum.Health > 0 then
                    local pos2D, onScreen = Camera:WorldToViewportPoint(p.Character.HumanoidRootPart.Position)
                    if onScreen then
                        local dist = (Vector2.new(pos2D.X, pos2D.Y) - center).Magnitude
                        if dist < minDist then minDist = dist; closest = p end
                    end
                end
            end
        end
        if closest then
            if closest ~= hoverLastPlayer then
                hoverLastPlayer = closest
                SetTargetPlayer(closest)
                UpdateInventoryUI(closest, true)
            else
                UpdateInventoryUI(closest, false)
            end
            SetUIVisibility(true)
        else
            SetUIVisibility(false); hoverLastPlayer = nil; cachedItems = {}
        end
    end

        pcall(function() HoverScreenGui:Destroy() end)
        getgenv().SW_hoverInvUpdate = nil
    end

do
    local GunChamsData = {Enabled=false, Color=Color3.fromRGB(0,230,255), Material=Enum.Material.Neon, Original={}}
    getgenv().SW_GunChamsData = GunChamsData
    local gunSafeWords = {"handguard", "handle", "firearm", "armature", "bolt", "mag", "stock"}
    local gunBodyWords = {"arm", "hand", "glove", "sleeve", "wrist", "finger", "thumb", "humanoidrootpart", "torso", "leg", "foot"}
    local function IsGunBodyPart(name)
        name = name:lower()
        for _, s in ipairs(gunSafeWords) do if name:find(s) then return false end end
        for _, b in ipairs(gunBodyWords) do if name:find(b) then return true end end
        return false
    end
    local function ApplyGunChams(part)
        if not GunChamsData.Original[part] then
            GunChamsData.Original[part] = {Material=part.Material, Color=part.Color}
        end
        part.Material = GunChamsData.Material
        part.Color = GunChamsData.Color
        if part:IsA("UnionOperation") then
            if not part.UsePartColor then part.UsePartColor = true end
        elseif part:IsA("MeshPart") then
            if part.TextureID ~= "" then part.TextureID = "" end
        end
        for _, child in ipairs(part:GetChildren()) do
            if child:IsA("SurfaceAppearance") or child:IsA("Decal") or child:IsA("Texture") then
                child:Destroy()
            elseif child:IsA("SpecialMesh") then
                if child.TextureId ~= "" then child.TextureId = "" end
                child.VertexColor = Vector3.new(1, 1, 1)
            end
        end
    end
    local function RestoreGunChams()
        for part, orig in pairs(GunChamsData.Original) do
            pcall(function()
                if part and part.Parent then
                    part.Material = orig.Material
                    part.Color = orig.Color
                end
            end)
        end
        table.clear(GunChamsData.Original)
    end
    getgenv().SW_RestoreGunChams = RestoreGunChams
    local gunChamsAcc = 0
    getgenv().SW_GunChamsUpdate = function(dt)
        if not GunChamsData.Enabled then
            if next(GunChamsData.Original) then RestoreGunChams() end
            return
        end
        gunChamsAcc = gunChamsAcc + (dt or 0.016)
        if gunChamsAcc < 0.1 then return end
        gunChamsAcc = 0
        local Flags = getgenv().SW_Flags
        GunChamsData.Color = Flags["combat_gun_chams_clr"] or GunChamsData.Color
        GunChamsData.Material = Enum.Material[Flags["combat_gun_chams_mat"] or "Neon"] or Enum.Material.Neon
        if LocalPlayer.Character then
            for _, child in ipairs(LocalPlayer.Character:GetChildren()) do
                if child:IsA("Tool") or (child:IsA("Model") and child:FindFirstChild("GunComponents")) then
                    for _, part in ipairs(child:GetDescendants()) do
                        if part:IsA("BasePart") and not IsGunBodyPart(part.Name) then ApplyGunChams(part) end
                    end
                end
            end
        end
        if workspace.CurrentCamera then
            for _, part in ipairs(workspace.CurrentCamera:GetDescendants()) do
                if part:IsA("BasePart") and not IsGunBodyPart(part.Name) then ApplyGunChams(part) end
            end
        end
    end
end

        elseif o:IsA("Model") or o:IsA("Folder") then
            for _,ch in ipairs(o:GetDescendants()) do hide(ch,k) end
        end
    end

        if UIS:IsKeyDown(Enum.KeyCode.S) then moveVec = moveVec + Vector3.new(0,0,1) end

    }
    local HIT_NAMES = { ["hit"]=true, ["headshothit"]=true, ["headshot"]=true }
    local selectedHitSoundId = ""
    local hitVolume = 1  -- 0-3, ставим через getgenv().SW_setHitVolume

    -- Проверяем, что звук принадлежит НАМ (Tool в нашем Character, Viewmodel, ViewmodelArms, Arms)
    local function isOurHitSound(snd)
        if not snd or not snd.Parent then return false end
        -- Прямой Tool в нашем Character
        local cur = snd.Parent
        local depth = 0
        while cur and depth < 8 do
            if cur == LocalPlayer.Character then return true end
            if cur.Name == "Viewmodel" or cur.Name == "ViewmodelArms" or cur.Name == "Arms" then return true end
            if cur:IsA("Tool") and cur.Parent == LocalPlayer.Character then return true end
            cur = cur.Parent; depth = depth + 1
        end
        -- Камера Viewmodel
        local cam = workspace.CurrentCamera
        if cam then
            local vm = cam:FindFirstChild("Viewmodel")
            if vm and snd:IsDescendantOf(vm) then return true end
            local vma = cam:FindFirstChild("ViewmodelArms")
            if vma and snd:IsDescendantOf(vma) then return true end
        end
        return false
    end

    -- Применяем громкость + SoundId к одному звуку
    local function applyToSound(snd)
        if not snd or not snd:IsA("Sound") then return end
        if not HIT_NAMES[string.lower(snd.Name)] then return end
        if selectedHitSoundId == "" then return end
        if not isOurHitSound(snd) then return end
        -- Единый SoundId
        if snd.SoundId ~= selectedHitSoundId then snd.SoundId = selectedHitSoundId end
        -- Единая громкость на всех частях тела
        if snd.Volume ~= hitVolume then snd.Volume = hitVolume end
        -- Отключаем затухание, чтобы звук из любой части играл одинаково
        if snd.RollOffMode ~= Enum.RollOffMode.Inverse then snd.RollOffMode = Enum.RollOffMode.Inverse end
        if snd.RollOffMinDistance ~= 10000 then snd.RollOffMinDistance = 10000 end
        if snd.RollOffMaxDistance ~= 10000 then snd.RollOffMaxDistance = 10000 end
        -- PlaybackSpeed выравниваем
        if snd.PlaybackSpeed ~= 1 then snd.PlaybackSpeed = 1 end
    end

    -- Применяем ко всем Sound в контейнере + подписываемся на новые
    local function applyToContainer(container)
        if not container then return end
        for _, d in ipairs(container:GetDescendants()) do
            if d:IsA("Sound") then applyToSound(d) end
        end
        container.DescendantAdded:Connect(function(d)
            if d:IsA("Sound") then task.wait(0.05); applyToSound(d) end
        end)
    end

    local function bindTool(tool) if tool then applyToContainer(tool) end end
    local function bindViewmodel(vm) if vm then applyToContainer(vm) end end

    local function setupCharacter(char)
        if not char then return end
        char.ChildAdded:Connect(function(child)
            if child:IsA("Tool") then bindTool(child) end

        end

        vmConn = cam.ChildAdded:Connect(function(child)
            if child.Name == "Viewmodel" or child.Name == "ViewmodelArms" or child.Name == "Arms" then
                bindViewmodel(child)
            end
        end)
    end

    if LocalPlayer.Character then setupCharacter(LocalPlayer.Character) end

    end)

                local vm = cam:FindFirstChild(cname)
                if vm then
                    for _, d in ipairs(vm:GetDescendants()) do
                        if d:IsA("Sound") then applyToSound(d) end
                    end
                end
            end
        end
    end

                end
            end
        end)

        getgenv().SW_CanvasGroup.Visible = true

-- ===================== UNLOAD =====================
function SW.unload()
    if not SW.Alive then return end
    SW.Alive=false
    for c in pairs(SW.Connections) do pcall(function() c:Disconnect() end) end
    table.clear(SW.Connections)
    pcall(getgenv().SW_stopAutoWalk)
    pcall(getgenv().SW_applyIR); pcall(getgenv().SW_applyFastImmediate)
    pcall(getgenv().SW_RestoreGunChams)
    pcall(getgenv().SW_setNoFog, false)
    pcall(getgenv().SW_restoreLighting)
    pcall(function()
        local ao = getgenv().SW_atmosOriginal
        Lighting.Ambient = ao.Ambient; Lighting.OutdoorAmbient = ao.OutdoorAmbient
        Lighting.FogColor = ao.FogColor
        Lighting.FogStart = ao.FogStart; Lighting.FogEnd = ao.FogEnd
    end)
    pcall(function() getgenv().SW_setVeg("Grass", false) end)
    pcall(function() getgenv().SW_setVeg("Leaves", false) end)
    if Camera then pcall(function() Camera.FieldOfView = 70 end) end
    for _,D in pairs(getgenv().SW_Esp.Players) do pcall(D.Destroy) end
    for o in pairs(SW.Owned) do pcall(function() o:Destroy() end) end
    table.clear(SW.Owned)
    pcall(function() getgenv().SW_silFovCircle:Remove() end)
    pcall(function() getgenv().SW_silTargetLine:Remove() end)
    if getgenv().SW_cleanupP2P then pcall(getgenv().SW_cleanupP2P) end
    if getgenv().SW_cleanupSkeleton then pcall(getgenv().SW_cleanupSkeleton) end
    if getgenv().Fanity_HitLogs_Cleanup then pcall(getgenv().Fanity_HitLogs_Cleanup) end
    if getgenv().SW_hitSoundCleanup then pcall(getgenv().SW_hitSoundCleanup) end
    local chLines = getgenv().SW_chLines
    if chLines then
        pcall(function() chLines[1]:Remove() end); pcall(function() chLines[2]:Remove() end)
        pcall(function() chLines[3]:Remove() end); pcall(function() chLines[4]:Remove() end)
    end
    if getgenv().Fanity_CrosshairTarget_Cleanup then pcall(getgenv().Fanity_CrosshairTarget_Cleanup) end
    if getgenv().SW_hoverInvCleanup then pcall(getgenv().SW_hoverInvCleanup) end

    local accXray, accWorld, accMove, accGunChams = 0, 0, 0, 0
    local THROTTLE = 0.033

    SW.connect(RunService.RenderStepped, function(dt)
        if not SW.Alive then return end
        getgenv().SW_fovApplyNow()
        getgenv().SW_updateCrosshair(dt)
        if getgenv().SW_targetHudUpdate then getgenv().SW_targetHudUpdate() end
        if getgenv().SW_espUpdate then getgenv().SW_espUpdate(dt) end
        if getgenv().SW_skeletonUpdate then getgenv().SW_skeletonUpdate() end
        if getgenv().SW_p2pUpdate then getgenv().SW_p2pUpdate(dt) end
        getgenv().SW_hoverInvUpdate(dt)
        getgenv().SW_aimlockUpdate()
        getgenv().SW_freecamUpdate(dt)
        getgenv().SW_triggerbotUpdate()
        getgenv().SW_silentUpdate()
    end)

    SW.connect(RunService.Heartbeat, function(dt)
        if not SW.Alive then return end
        local Flags = getgenv().SW_Flags
        if Flags["vis_tracer"] and #getgenv().SW_pendingTracers > 0 then
            local budget = 4
            while #getgenv().SW_pendingTracers > 0 and budget > 0 do
                budget = budget - 1
                getgenv().SW_spawnTracerBeam(table.remove(getgenv().SW_pendingTracers, 1))
            end
        end
        if getgenv().SW_worldEspUpdate then getgenv().SW_worldEspUpdate(dt) end

        accXray = accXray + dt
        if accXray >= THROTTLE then
            accXray = 0
            if getgenv().SW_xrayUpdate then getgenv().SW_xrayUpdate(dt) end
        end

        accWorld = accWorld + dt
        if accWorld >= THROTTLE then
            accWorld = 0
            if getgenv().SW_worldUpdate then getgenv().SW_worldUpdate(dt) end
        end

        accMove = accMove + dt
        if accMove >= THROTTLE then
            accMove = 0
            if getgenv().SW_moveUpdate then getgenv().SW_moveUpdate(dt) end
        end

        accGunChams = accGunChams + dt
        if accGunChams >= THROTTLE then
            accGunChams = 0
            if getgenv().SW_GunChamsUpdate then getgenv().SW_GunChamsUpdate(dt) end
        end

        if Flags["combat_instantreload"] then
            irAcc = irAcc + dt
            if irAcc >= 2 then
                irAcc = 0
                pcall(getgenv().SW_applyIR)
            end
        end
    end)
end
