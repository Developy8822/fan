local function MakeDraggable(gui)
    local dragging, dragInput, dragStart, startPos
    gui.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = true; dragStart = input.Position; startPos = gui.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then dragging = false end
            end)
        end
    end)
    gui.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement then dragInput = input end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if input == dragInput and dragging then
            local delta = input.Position - dragStart
            gui.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end)
end
getgenv().IMP_MakeDraggable = MakeDraggable

-- color helpers
local function color3ToTable(c)
    if type(c) == "table" then return c end
    if typeof(c) == "Color3" then return {c.R, c.G, c.B} end
    return {1, 1, 1}
end
local function tableToColor3(t)
    if type(t) == "Color3" then return t end
    if type(t) == "table" and t[1] and t[2] and t[3] then return Color3.new(t[1], t[2], t[3]) end
    return Color3.new(1, 1, 1)
end

local ConfigFolder = "ImperiumConfigs"
if isfolder and not isfolder(ConfigFolder) then makefolder(ConfigFolder) end
getgenv().IMP_CONFIG_FOLDER = ConfigFolder

function Library:GetConfigs()
    local list = {}
    if listfiles then
        for _, file in pairs(listfiles(ConfigFolder)) do
            if file:match(".json$") then
                table.insert(list, file:match("([^/\\]+)%.json$"))
            end
        end
    end
    return #list > 0 and list or {"No Configs"}
end

function Library:SaveConfig(name)
    if not name or name == "" or name == "No Configs" then return end
    local m, w, b = self.Windows.Main, self.Windows.Watermark, self.Windows.Binds
    local positions = {
        Main = {m.Position.X.Scale, m.Position.X.Offset, m.Position.Y.Scale, m.Position.Y.Offset},
        WM = {w.Position.X.Scale, w.Position.X.Offset, w.Position.Y.Scale, w.Position.Y.Offset},
        Binds = {b.Position.X.Scale, b.Position.X.Offset, b.Position.Y.Scale, b.Position.Y.Offset}
    }
    local data = {Flags = self.Flags, Positions = positions}
    if writefile then
        writefile(ConfigFolder .. "/" .. name .. ".json", HttpService:JSONEncode(data))
    end
end

function Library:LoadConfig(name)
    if not name or name == "" or name == "No Configs" then return end
    local path = ConfigFolder .. "/" .. name .. ".json"
    if readfile and isfile and isfile(path) then
        local s, d = pcall(function() return HttpService:JSONDecode(readfile(path)) end)
        if s and d then
            if d.Flags then
                for k, v in pairs(d.Flags) do
                    if self.Setters[k] then pcall(function() self.Setters[k](v) end) end
                end
            end
            if d.Positions then
                local p = d.Positions
                if p.Main then self.Windows.Main.Position = UDim2.new(unpack(p.Main)) end
                if p.WM then self.Windows.Watermark.Position = UDim2.new(unpack(p.WM)) end
                if p.Binds then self.Windows.Binds.Position = UDim2.new(unpack(p.Binds)) end
            end
        end
    end
end

function Library:CreateConfigManager(Tab, Column)
    local typedName = ""
    local selectedName = ""
    Tab:AddTextBox(Column, "Config Name", "Enter name to save...", function(v) typedName = v end)
    Tab:AddDropdown(Column, "Select Config", self:GetConfigs(), "No Configs", function(v) selectedName = v end)
    Tab:AddButton(Column, "Save", function()
        local n = (typedName ~= "") and typedName or selectedName
        if n and n ~= "" and n ~= "No Configs" then
            self:SaveConfig(n)
            local newConfigs = self:GetConfigs()
            if self.Dropdowns["Select Config"] then self.Dropdowns["Select Config"](newConfigs) end
            if self.Setters["Select Config"] then self.Setters["Select Config"](n) end
            if self.Setters["Config Name"] then self.Setters["Config Name"]("") end
            typedName = ""
        end
    end)
    Tab:AddButton(Column, "Load", function() self:LoadConfig(selectedName) end)
    Tab:AddButton(Column, "Delete", function()
        if selectedName and selectedName ~= "No Configs" then
            local path = ConfigFolder .. "/" .. selectedName .. ".json"
            if delfile and isfile and isfile(path) then delfile(path) end
            local currentConfigs = self:GetConfigs()
            if self.Dropdowns["Select Config"] then self.Dropdowns["Select Config"](currentConfigs) end
            if self.Setters["Select Config"] then self.Setters["Select Config"](currentConfigs[1]) end
        end
    end)
    Tab:AddButton(Column, "Refresh List", function()
        local currentConfigs = self:GetConfigs()
        if self.Dropdowns["Select Config"] then self.Dropdowns["Select Config"](currentConfigs) end
        if self.Setters["Select Config"] then self.Setters["Select Config"](currentConfigs[1]) end
    end)
end

-- Triggerbot
do
    local trigShooting = false
    local function trigHasTarget()
        local m = UserInputService:GetMouseLocation()
        local mx, my = m.X, m.Y
        local fov = Flags["Triggerbot FOV"] or 150
        for _, plr in ipairs(Players:GetPlayers()) do
            if plr ~= Player then
                local char = plr.Character
                if char then
                    local hum = char:FindFirstChildOfClass("Humanoid")
                    local part = char:FindFirstChild("Head")
                    if hum and hum.Health > 0 and part then
                        local sp, onScreen = Camera:WorldToViewportPoint(part.Position)
                        if onScreen then
                            local dx = sp.X - mx
                            local dy = sp.Y - my
                            if dx * dx + dy * dy <= fov * fov then return true end
                        end
                    end
                end
            end
        end
        return false
    end
    getgenv().IMP_triggerbotUpdate = function()
        if not Flags["Triggerbot"] then
            if trigShooting then
                trigShooting = false
                pcall(function()
                    if mouse1release then mouse1release()
                    else VirtualInputManager:SendMouseButtonEvent(0, 0, 0, false, game, 0) end
                end)
            end
            return
        end
        if trigHasTarget() then
            if not trigShooting then
                trigShooting = true
                pcall(function()
                    if mouse1press then mouse1press()
                    else VirtualInputManager:SendMouseButtonEvent(0, 0, 0, true, game, 0) end
                end)
            end
        else
            if trigShooting then
                trigShooting = false
                pcall(function()
                    if mouse1release then mouse1release()
                    else VirtualInputManager:SendMouseButtonEvent(0, 0, 0, false, game, 0) end
                end)
            end
        end
    end

    getgenv().IMP_applyIR = function()
        if not ensureItemData() then return end
        for name, item in pairs(itemDataRef.Items) do
            if type(item) == "table" and type(item.ReloadInfo) == "table" then
                if Flags["Instant Reload"] then
                    if not savedReload[name] then
                        savedReload[name] = {RD = item.ReloadInfo.ReloadDuration, RT = item.ReloadInfo.ReturnsMagTime, GT = item.ReloadInfo.GetsMagTime}
                    end
                    item.ReloadInfo.ReloadDuration = 0.01
                    item.ReloadInfo.ReturnsMagTime = 0.01
                    item.ReloadInfo.GetsMagTime = 0.01
                elseif savedReload[name] then
                    item.ReloadInfo.ReloadDuration = savedReload[name].RD
                    item.ReloadInfo.ReturnsMagTime = savedReload[name].RT
                    item.ReloadInfo.GetsMagTime = savedReload[name].GT
                end
            end
        end
    end

    local function applyFastNow()
        if not ensureItemData() then return false end
        local rates = {MP5 = 420, ["Semi-Automatic Pistol"] = 330, ["Old Revolver"] = 270, ["Double Barrel Shotgun"] = 150, ["Badger Rifle"] = 400, ["Nomad Rifle"] = 400, Thompson = 400, ["Rocket Launcher"] = 60}
        local times = {Rock = {0.34, 0.5, 0.6}, ["Stone Hatchet"] = {0.28, 0.5, 0.25}, ["Metal Hatchet"] = {0.28, 0.5, 0.25}, ["Stone Pickaxe"] = {0.29, 0.47, 0.25}, ["Metal Pickaxe"] = {0.29, 0.47, 0.25}, ["Wooden Spear"] = {0.2, 0.57, 0.25}}
        local mult = Flags["Fast Tools Mult"] or 8
        for n, base in pairs(rates) do
            local it = itemDataRef.Items[n]
            if it then
                if not originalTools[n] then originalTools[n] = {FR = it.FireRate} end
                it.FireRate = Flags["Fast Tools"] and math.clamp(base * mult, 1, 8000) or originalTools[n].FR
            end
        end
        for n, base in pairs(times) do
            local it = itemDataRef.Items[n]
            if it and it.ToolInfo then
                if not originalTools[n] then originalTools[n] = {} end
                if not originalTools[n].T then
                    originalTools[n].T = {it.ToolInfo.HitTime, it.ToolInfo.PlayTime, it.ToolInfo.WaitTime}
                end
                local v = originalTools[n].T
                it.ToolInfo.HitTime = Flags["Fast Tools"] and math.max(0.01, base[1] / mult) or v[1]
                it.ToolInfo.PlayTime = Flags["Fast Tools"] and math.max(0.05, base[2] / mult) or v[2]
                it.ToolInfo.WaitTime = Flags["Fast Tools"] and math.max(0.01, base[3] / mult) or v[3]
            end
        end
        return true
    end

    local debounceScheduled = false
    getgenv().IMP_applyFast = function()
        if debounceScheduled then return end
        debounceScheduled = true
        task.delay(0.05, function()
            applyFastNow()
            debounceScheduled = false
        end)
    end

                            end
                        end
                    end
                end)

        end
    end)

    local gunBodyWords = {"arm", "hand", "glove", "sleeve", "wrist", "finger", "thumb", "humanoidrootpart", "torso", "leg", "foot"}

            GunChamsData.Original[part] = {Material = part.Material, Color = part.Color}
        end
        part.Material = GunChamsData.Material
        part.Color = GunChamsData.Color
        if part:IsA("UnionOperation") then
            if not part.UsePartColor then part.UsePartColor = true end
        elseif part:IsA("MeshPart") then
            if part.TextureID ~= "" then part.TextureID = "" end
        end
        for _, child in ipairs(part:GetChildren()) do
            if child:IsA("SurfaceAppearance") or child:IsA("Decal") or child:IsA("Texture") then
                child:Destroy()
            elseif child:IsA("SpecialMesh") then
                if child.TextureId ~= "" then child.TextureId = "" end
                child.VertexColor = Vector3.new(1, 1, 1)
            end
        end
    end

    local function RestoreGunChams()
        for part, orig in pairs(GunChamsData.Original) do
            pcall(function()
                if part and part.Parent then
                    part.Material = orig.Material
                    part.Color = orig.Color
                end
            end)
        end
        table.clear(GunChamsData.Original)
    end
    getgenv().IMP_RestoreGunChams = RestoreGunChams

    local acc = 0
    getgenv().IMP_GunChamsUpdate = function(dt)
        if not GunChamsData.Enabled then
            if next(GunChamsData.Original) then RestoreGunChams() end
            return
        end
        acc = acc + (dt or 0.016)
        if acc < 0.1 then return end
        acc = 0
        local c = Flags["Gun Chams Color_Color"] or {0, 1, 1}
        GunChamsData.Color = Color3.new(c[1], c[2], c[3])
        GunChamsData.Material = Enum.Material[Flags["Gun Chams Material"] or "Neon"] or Enum.Material.Neon
        if Player.Character then
            for _, child in ipairs(Player.Character:GetChildren()) do
                if child:IsA("Tool") or (child:IsA("Model") and child:FindFirstChild("GunComponents")) then
                    for _, part in ipairs(child:GetDescendants()) do
                        if part:IsA("BasePart") and not IsGunBodyPart(part.Name) then ApplyGunChams(part) end
                    end
                end
            end
        end
        if Workspace.CurrentCamera then
            for _, part in ipairs(Workspace.CurrentCamera:GetDescendants()) do
                if part:IsA("BasePart") and not IsGunBodyPart(part.Name) then ApplyGunChams(part) end
            end
        end
    end

    }
    local HIT_NAMES = {["hit"] = true, ["headshothit"] = true, ["headshot"] = true}
    local selectedHitSoundId = ""

    local function isOurHitSound(snd)
        if not snd or not snd.Parent then return false end
        local parent = snd.Parent
        local depth = 0
        while parent and parent ~= Workspace and depth < 6 do
            if parent:IsA("Tool") and parent.Parent == Player.Character then return true end
            if parent.Name == "Viewmodel" or parent.Name == "ViewmodelArms" then return true end
            parent = parent.Parent
            depth = depth + 1
        end
        local vm = Camera:FindFirstChild("Viewmodel")
        if vm and snd:IsDescendantOf(vm) then return true end
        return false
    end

    local function applyToSound(snd)
        if not snd or not snd:IsA("Sound") then return end
        if not HIT_NAMES[string.lower(snd.Name)] then return end
        if selectedHitSoundId == "" then return end
        if isOurHitSound(snd) then snd.SoundId = selectedHitSoundId end
    end

    local function applyToContainer(container)
        if not container then return end
        for _, d in ipairs(container:GetDescendants()) do
            if d:IsA("Sound") then applyToSound(d) end
        end
        container.DescendantAdded:Connect(function(d)
            if d:IsA("Sound") then
                task.wait(0.05)
                applyToSound(d)
            end
        end)
    end

    local function bindTool(tool) if tool then applyToContainer(tool) end end
    local function bindViewmodel(vm) if vm then applyToContainer(vm) end end

    local function setupCharacter(char)
        if not char then return end
        char.ChildAdded:Connect(function(child)
            if child:IsA("Tool") then bindTool(child) end
        end)
        for _, child in ipairs(char:GetChildren()) do
            if child:IsA("Tool") then bindTool(child) end
        end
    end

    local vmConn = nil
    local function setupViewmodel()
        if vmConn then pcall(function() vmConn:Disconnect() end); vmConn = nil end
        local cam = Workspace.CurrentCamera
        if not cam then return end
        local vm = cam:FindFirstChild("Viewmodel")
        if vm then bindViewmodel(vm) end
        vmConn = cam.ChildAdded:Connect(function(child)
            if child.Name == "Viewmodel" or child.Name == "ViewmodelArms" then bindViewmodel(child) end
        end)
    end

    if Player.Character then setupCharacter(Player.Character) end
    Player.CharacterAdded:Connect(function(ch)
        task.wait(0.5)
        setupCharacter(ch)
        setupViewmodel()
    end)
    setupViewmodel()

            end
        end
    end

                            local rX = math.floor(rootPos.X)
                            local tY = math.floor(topPos.Y)
                            local bY = math.floor(botPos.Y)
                            local tl_X = math.floor(rX - width / 2)
                            local tl_Y = tY

                            d.BoxOutline.Visible = true
                            d.BoxOutline.Size = Vector2.new(width, height)
                            d.BoxOutline.Position = Vector2.new(tl_X, tl_Y)
                            d.Box.Visible = true

                            d.Box.Position = Vector2.new(tl_X, tl_Y)

                                d.NameText.Position = Vector2.new(rX, currentTextY)
                            else
                                d.NameText.Visible = false
                            end

                            if showDist then
                                d.DistanceText.Visible = true
                                local dist = (camPos - root.Position).Magnitude
                                d.DistanceText.Text = math.floor(dist) .. "m"
                                d.DistanceText.Position = Vector2.new(rX, bY + 2)
                            else
                                d.DistanceText.Visible = false
                            end

                            if showSkel and not isTiny then
                                local rig = hum.RigType == Enum.HumanoidRigType.R15 and R15_BONES or R6_BONES
                                for i = 1, #rig do
                                    local pair = rig[i]
                                    local line = d.Skeleton[i]

                                    local p1 = d.Parts[n1]
                                    if not p1 then p1 = char:FindFirstChild(n1); if p1 then d.Parts[n1] = p1 end end
                                    local p2 = d.Parts[n2]

    local function RegisterItemImage(name, url)
        if not name or type(name) ~= "string" or #name < 2 then return end
        local asset = toAsset(url)
        if asset then
            IMAGES[name] = asset

    task.spawn(function()
        local function DeepScan(t, depth)
            if depth > 4 or type(t) ~= "table" then return end
            for k, v in pairs(t) do
                if type(v) == "table" then
                    local name = v.Name or v.ItemName or (type(k) == "string" and k)
                    local img = v.Image or v.Icon or v.Texture or v.IconId or v.ImageId
                    if name and img and isGoodImage(tostring(img)) then
                        RegisterItemImage(tostring(name), tostring(img))
                    end
                    DeepScan(v, depth + 1)
                elseif type(v) == "string" and isGoodImage(v) and type(k) == "string" and #k > 1 then
                    RegisterItemImage(k, v)
                end
            end
        end
        pcall(function()
            for _, desc in ipairs(ReplicatedStorage:GetDescendants()) do
                if desc:IsA("ModuleScript") then
                    pcall(function()
                        local res = require(desc)
                        if type(res) == "table" then DeepScan(res, 1) end
                    end)
                elseif desc:IsA("StringValue") and (desc.Name == "Icon" or desc.Name == "Image") and isGoodImage(desc.Value) and desc.Parent then
                    RegisterItemImage(desc.Parent.Name, desc.Value)
                end
            end
        end)
    end)

    local __imgCache = {}
    local function GetItemImage(itemName)
        if not itemName then return nil end
        if __imgCache[itemName] ~= nil then return __imgCache[itemName] or nil end
        local result = nil
        if IMAGES[itemName] then
            result = IMAGES[itemName]
        else
            local lower = string.lower(itemName)
            if IMAGES[lower] then
                result = IMAGES[lower]
            else
                local alias = ALIASES[lower] or ALIASES[itemName]
                if alias then
                    result = IMAGES[alias] or IMAGES[string.lower(alias)]
                end
            end
        end
        __imgCache[itemName] = result or false
        return result
    end

    local function EnableSmoothDrag(frame, dragHandle)
        local dragging = false
        local dragStart = Vector2.new()
        local startPos = Vector2.new()
        local targetPos = Vector2.new(frame.Position.X.Offset, frame.Position.Y.Offset)
        local activeConn = nil
        local function startLoop()
            if activeConn then return end
            activeConn = RunService.RenderStepped:Connect(function(dt)
                if not frame.Visible then return end
                local cur = Vector2.new(frame.Position.X.Offset, frame.Position.Y.Offset)
                local lerped = cur:Lerp(targetPos, math.clamp(dt * 24, 0, 1))
                frame.Position = UDim2.new(frame.Position.X.Scale, math.round(lerped.X), frame.Position.Y.Scale, math.round(lerped.Y))
                if not dragging and (lerped - targetPos).Magnitude < 0.5 then
                    pcall(function() activeConn:Disconnect() end)
                    activeConn = nil
                end
            end)
        end
        dragHandle.InputBegan:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                dragging = true
                dragStart = Vector2.new(input.Position.X, input.Position.Y)
                startPos = Vector2.new(frame.Position.X.Offset, frame.Position.Y.Offset)
                startLoop()
            end
        end)
        UserInputService.InputEnded:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                dragging = false
            end
        end)
        UserInputService.InputChanged:Connect(function(input)
            if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
                local delta = Vector2.new(input.Position.X, input.Position.Y) - dragStart
                targetPos = startPos + delta
            end
        end)
    end

    local SKIP = {[""] = true, [" "] = true, Empty = true, ["0"] = true, None = true}

    local function AreIdentical(t1, t2)
        if not t1 or not t2 then return false end
        for k, v in pairs(t1) do if t2[k] ~= v then return false end end
        for k, v in pairs(t2) do if t1[k] ~= v then return false end end
        return true
    end

    local function UpdateInventoryUI(player, force)
        local dataFolder = player:FindFirstChild("Data")
        if not dataFolder then return end
        local currentItems = {}
        for _, slot in ipairs(dataFolder:GetChildren()) do
            local nameObj = slot:FindFirstChild("ItemName") or slot:FindFirstChild("Item") or slot:FindFirstChild("Name")
            local name
            if nameObj and nameObj:IsA("StringValue") then
                name = nameObj.Value
            elseif slot:IsA("StringValue") then
                name = slot.Value
            end
            if name then name = name:match("^%s*(.-)%s*$") end
            if name and not SKIP[name] and #name > 0 then
                local add = 1
                local amt = slot:FindFirstChild("Amount") or slot:FindFirstChild("Count") or slot:FindFirstChild("Quantity") or slot:FindFirstChild("Stack")
                if amt and (amt:IsA("IntValue") or amt:IsA("NumberValue")) and amt.Value > 0 then
                    add = amt.Value
                end
                currentItems[name] = (currentItems[name] or 0) + add
            end
        end
        if not force and AreIdentical(cachedItems, currentItems) then return end
        cachedItems = currentItems
        for _, child in pairs(ItemScroll:GetChildren()) do
            if child:IsA("Frame") then child:Destroy() end
        end
        local totalItems = 0
        for itemName, count in pairs(currentItems) do
            CreateItemSlot(itemName, count)
            totalItems = totalItems + 1
        end
        SubText.Text = string.format("ITEMS COUNT: %d", totalItems)
        ItemScroll.CanvasSize = UDim2.new(0, 0, 0, math.ceil(math.max(1, totalItems) / 3) * 100)
    end

    local function SetTargetPlayer(player)
        TitleText.Text = string.format("<b>%s</b> <font color=\"#8A8F9E\">[imperium]</font>", player.DisplayName or player.Name)
        task.spawn(function()
            local thumb = Players:GetUserThumbnailAsync(player.UserId, Enum.ThumbnailType.HeadShot, Enum.ThumbnailSize.Size100x100)
            if thumb and Avatar then Avatar.Image = thumb end
        end)
    end

    local hoverAcc = 0
    getgenv().IMP_hoverInvUpdate = function(dt)
        if not Flags["Hover Inventory"] then
            if hoverLastPlayer then
                SetUIVisibility(false)
                hoverLastPlayer = nil
                cachedItems = {}
            end
            return
        end
        hoverAcc = hoverAcc + (dt or 0.016)
        if hoverAcc < 0.1 then return end
        hoverAcc = 0
        local center = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
        local maxDist = Flags["Hover Inv Dist"] or 150
        local effectiveMaxDist = hoverLastPlayer and (maxDist + 40) or maxDist
        local closest, minDist = nil, effectiveMaxDist
        for _, p in pairs(Players:GetPlayers()) do
            if p ~= Player and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
                local hum = p.Character:FindFirstChild("Humanoid")
                if hum and hum.Health > 0 then
                    local pos2D, onScreen = Camera:WorldToViewportPoint(p.Character.HumanoidRootPart.Position)
                    if onScreen then
                        local dist = (Vector2.new(pos2D.X, pos2D.Y) - center).Magnitude
                        if dist < minDist then
                            minDist = dist
                            closest = p
                        end
                    end
                end
            end
        end
        if closest then
            if closest ~= hoverLastPlayer then
                hoverLastPlayer = closest
                SetTargetPlayer(closest)
                UpdateInventoryUI(closest, true)
            else
                UpdateInventoryUI(closest, false)
            end
            SetUIVisibility(true)
        else
            SetUIVisibility(false)
            hoverLastPlayer = nil
            cachedItems = {}
        end
    end

        pcall(function() HoverScreenGui:Destroy() end)
        getgenv().IMP_hoverInvUpdate = nil
    end

    local function startAutoWalk()
        if autowalkState.walking then return end
        autowalkState.walking = true
        autowalkState.conn = RunService.Heartbeat:Connect(function()
            if not autowalkState.walking or not Flags["Auto Walk"] then
                stopAutoWalk()
                return
            end
            pcall(function()
                VirtualInputManager:SendKeyEvent(true, Enum.KeyCode.W, false, game)
                if Flags["Auto Run"] then
                    VirtualInputManager:SendKeyEvent(true, Enum.KeyCode.LeftShift, false, game)
                end
            end)
        end)
    end
    getgenv().IMP_stopAutoWalk = stopAutoWalk
    getgenv().IMP_startAutoWalk = startAutoWalk
    Player.CharacterAdded:Connect(function() stopAutoWalk() end)

    local fcPos
    local fcRot = Vector2.new(0, 0)
    local fcActiveLast = false
    getgenv().IMP_freecamUpdate = function(dt)
        if not Flags["Freecam"] then
            if fcActiveLast then
                Camera.CameraType = Enum.CameraType.Custom
                Camera.CameraSubject = Player.Character and Player.Character:FindFirstChildOfClass("Humanoid")
                fcActiveLast = false
            end
            return
        end
        if not fcActiveLast then
            fcPos = Camera.CFrame.Position
            fcRot = Vector2.new(0, 0)
        end
        fcActiveLast = true
        Camera.CameraType = Enum.CameraType.Scriptable
        local speed = Flags["Freecam Speed"] or 50
        if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then speed = speed * 2.5 end
        local moveVec = Vector3.new()
        if UserInputService:IsKeyDown(Enum.KeyCode.W) then moveVec = moveVec + Vector3.new(0, 0, -1) end
        if UserInputService:IsKeyDown(Enum.KeyCode.S) then moveVec = moveVec + Vector3.new(0, 0, 1) end
        if UserInputService:IsKeyDown(Enum.KeyCode.A) then moveVec = moveVec + Vector3.new(-1, 0, 0) end
        if UserInputService:IsKeyDown(Enum.KeyCode.D) then moveVec = moveVec + Vector3.new(1, 0, 0) end
        if UserInputService:IsKeyDown(Enum.KeyCode.E) then moveVec = moveVec + Vector3.new(0, 1, 0) end
        if UserInputService:IsKeyDown(Enum.KeyCode.Q) then moveVec = moveVec + Vector3.new(0, -1, 0) end
        if UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton2) then
            local delta = UserInputService:GetMouseDelta()
            fcRot = fcRot - Vector2.new(delta.X * 0.003, delta.Y * 0.003)
            fcRot = Vector2.new(fcRot.X, math.clamp(fcRot.Y, -1.57, 1.57))
        end
        local rotCF = CFrame.Angles(0, fcRot.X, 0) * CFrame.Angles(fcRot.Y, 0, 0)
        fcPos = fcPos + (rotCF * moveVec) * (speed * dt)
        Camera.CFrame = CFrame.new(fcPos) * rotCF
    end

    local fovApplied = false
    local function fovApplyNow()
        if not Camera then return end
        if Flags["FOV Changer"] then
            local v = Flags["FOV Value"] or 90
            if Camera.FieldOfView ~= v then Camera.FieldOfView = v end
            fovApplied = true
        else
            if fovApplied then
                Camera.FieldOfView = 70
                fovApplied = false
            end
        end
    end
    getgenv().IMP_fovApplyNow = fovApplyNow
    getgenv().IMP_applyFOV = fovApplyNow
    Camera:GetPropertyChangedSignal("FieldOfView"):Connect(function()
        if Flags["FOV Changer"] and Camera.FieldOfView ~= (Flags["FOV Value"] or 90) then
            Camera.FieldOfView = Flags["FOV Value"] or 90
        end
    end)
    Player.CharacterAdded:Connect(function()
        task.wait(0.1)
        if Flags["FOV Changer"] then fovApplyNow() end
    end)

    local moveAcc = 0
    getgenv().IMP_moveUpdate = function(dt)
        if Flags["Auto Walk"] then
            if not autowalkState.walking then startAutoWalk() end
        else
            if autowalkState.walking then stopAutoWalk() end
        end
        moveAcc = moveAcc + (dt or 0.016)
        if moveAcc >= 0.1 then
            moveAcc = 0
            local hum = Player.Character and Player.Character:FindFirstChildOfClass("Humanoid")
            if Flags["Speed Hack"] and hum and hum.WalkSpeed ~= Flags["Walk Speed"] then
                pcall(function() hum.WalkSpeed = Flags["Walk Speed"] end)
            end
        end
        if Flags["NoClip"] then
            local ch = Player.Character
            if ch then
                for _, o in ipairs(ch:GetDescendants()) do
                    if o:IsA("BasePart") and o.CanCollide then o.CanCollide = false end
                end
            end
        end
    end
end

do
    local savedLighting
    local fbCC = Instance.new("ColorCorrectionEffect")
    fbCC.Name = "IMP_FB"
    fbCC.Enabled = false
    fbCC.Parent = Lighting

    local noFogState = {Active = false, FogStart = Lighting.FogStart, FogEnd = Lighting.FogEnd, Atmos = nil}
    local function setNoFog(state)
        if state then
            if not noFogState.Active then
                noFogState.FogStart = Lighting.FogStart
                noFogState.FogEnd = Lighting.FogEnd
                noFogState.Atmos = Lighting:FindFirstChildOfClass("Atmosphere")
                if noFogState.Atmos then noFogState.Atmos.Parent = nil end
                noFogState.Active = true
            end
            Lighting.FogStart = 0
            Lighting.FogEnd = 100000
        else
            if noFogState.Active then
                Lighting.FogStart = noFogState.FogStart
                Lighting.FogEnd = noFogState.FogEnd
                if noFogState.Atmos then pcall(function() noFogState.Atmos.Parent = Lighting end) end
                noFogState.Atmos = nil
                noFogState.Active = false
            end
        end
    end
    getgenv().IMP_setNoFog = setNoFog

    local customAtmosData = {Color = Color3.fromRGB(20, 20, 30)}
    getgenv().IMP_customAtmosData = customAtmosData

    local atmosOriginal = {
        Ambient = Lighting.Ambient,
        OutdoorAmbient = Lighting.OutdoorAmbient,
        FogColor = Lighting.FogColor,
        FogStart = Lighting.FogStart,
        FogEnd = Lighting.FogEnd,
        ClockTime = Lighting.ClockTime,
        GlobalShadows = Lighting.GlobalShadows
    }
    getgenv().IMP_atmosOriginal = atmosOriginal

    local function applySimpleAtmos()
        if not Flags["Custom Atmos"] then
            pcall(function()
                Lighting.Ambient = atmosOriginal.Ambient
                Lighting.OutdoorAmbient = atmosOriginal.OutdoorAmbient
                Lighting.FogColor = atmosOriginal.FogColor
                if not noFogState.Active then
                    Lighting.FogStart = atmosOriginal.FogStart
                    Lighting.FogEnd = atmosOriginal.FogEnd
                end
                Lighting.GlobalShadows = atmosOriginal.GlobalShadows
            end)
            return
        end
        local cTab = Flags["Atmos Color_Color"] or {0.1, 0.1, 0.1}
        local c = Color3.new(cTab[1], cTab[2], cTab[3])
        pcall(function()
            Lighting.Ambient = c
            Lighting.OutdoorAmbient = c
            Lighting.FogColor = c
            if not noFogState.Active then
                Lighting.FogStart = 0
                Lighting.FogEnd = 100000
            end
            Lighting.GlobalShadows = false
        end)
        local atm = Lighting:FindFirstChildOfClass("Atmosphere")
        if atm then
            atm.Color = c
            atm.Decay = c
            atm.Density = 0.1
            atm.Haze = 0
            atm.Glare = 0
        end
    end
    getgenv().IMP_applySimpleAtmos = applySimpleAtmos

    local function saveLighting()
        if not savedLighting then
            savedLighting = {
                Ambient = Lighting.Ambient,
                OutdoorAmbient = Lighting.OutdoorAmbient,
                GlobalShadows = Lighting.GlobalShadows,
                Exposure = Lighting.ExposureCompensation
            }
        end
    end
    local function restoreLighting()
        if savedLighting then
            Lighting.Ambient = savedLighting.Ambient
            Lighting.OutdoorAmbient = savedLighting.OutdoorAmbient
            Lighting.GlobalShadows = savedLighting.GlobalShadows
            Lighting.ExposureCompensation = savedLighting.Exposure
            fbCC.Enabled = false
            fbCC.Brightness = 0
            savedLighting = nil
        end
    end
    getgenv().IMP_restoreLighting = restoreLighting

    local timeApplying = false
    local function forceTime()
        if Flags["Time Changer"] and not timeApplying then
            if math.abs(Lighting.ClockTime - Flags["Hour"]) > 0.05 then
                timeApplying = true
                Lighting.ClockTime = Flags["Hour"]
                timeApplying = false
            end
        end
    end
    getgenv().IMP_forceTime = forceTime
    getgenv().IMP_setTimeApplying = function(v) timeApplying = v end
    Lighting:GetPropertyChangedSignal("ClockTime"):Connect(forceTime)

    local VS = {Grass = false, Leaves = false}
    local Saved = {}
    local DecOrig = nil
    local function cat(o)
        local n = string.lower(o.Name)
        if n == "leaves" or n == "snow leaves" or n == "leaf" or n == "foliage" or n:find("leaves", 1, true) then return "Leaves" end
        if n == "grass" or n == "tall grass" or n == "grasspatch" or n:find("grass", 1, true) then return "Grass" end
    end
    local tree = {Tree = true, Wood = true}
    local function insideTree(p)
        local par = p.Parent
        while par and par ~= Workspace do
            if par:IsA("Model") and tree[par.Name] then return true end
            par = par.Parent
        end
        return false
    end
    local function hide(p, kind)
        if not p:IsA("BasePart") then return end
        if kind == "Grass" and insideTree(p) then return end
        local e = Saved[p]
        if not e then
            e = {T = p.LocalTransparencyModifier, K = {}}
            Saved[p] = e
        end
        e.K[kind] = true
        p.LocalTransparencyModifier = 1
    end
    local function visit(o)
        local k = cat(o)
        if not k and o:IsA("BasePart") then
            local par = o.Parent
            while par and par ~= Workspace do
                k = cat(par)
                if k then break end
                par = par.Parent
            end
        end
        if not k or not VS[k] then return end
        if o:IsA("BasePart") then
            hide(o, k)
        elseif o:IsA("Model") or o:IsA("Folder") then
            for _, ch in ipairs(o:GetDescendants()) do hide(ch, k) end
        end
    end
    getgenv().IMP_setVeg = function(kind, enabled)
        VS[kind] = enabled
        if enabled then
            for _, o in ipairs(Workspace:GetDescendants()) do visit(o) end
        else
            for p, e in pairs(Saved) do
                e.K[kind] = nil
                if not next(e.K) then
                    pcall(function() p.LocalTransparencyModifier = e.T end)
                    Saved[p] = nil
                end
            end
        end
        if kind == "Grass" then
            local t = Workspace.Terrain
            if enabled and DecOrig == nil then
                local ok, v = pcall(function() return t.Decoration end)
                if not ok and typeof(gethiddenproperty) == "function" then
                    ok, v = pcall(gethiddenproperty, t, "Decoration")
                end
                if ok and type(v) == "boolean" then DecOrig = v end
            end
            if DecOrig ~= nil then
                local tgt = DecOrig
                if enabled then tgt = false end
                local ok = pcall(function() t.Decoration = tgt end)
                if not ok and typeof(sethiddenproperty) == "function" then
                    ok = pcall(sethiddenproperty, t, "Decoration", tgt)
                end
            end
        end
    end
    Workspace.DescendantAdded:Connect(visit)

    local xrayApplied = {}
    local xrayAcc = 0
    local function xrayUpdate(dt)
        if not Flags["X-Ray"] then
            if next(xrayApplied) then
                for part, orig in pairs(xrayApplied) do
                    pcall(function() part.LocalTransparencyModifier = orig end)
                end
                table.clear(xrayApplied)
            end
            return
        end
        xrayAcc = xrayAcc + (dt or 0.016)
        if xrayAcc < 0.5 then return end
        xrayAcc = 0
        local pbs = Workspace:FindFirstChild("PlayerBuiltStructures")
        local folder = pbs and pbs:FindFirstChild("Structures")
        if folder then
            for _, part in ipairs(folder:GetDescendants()) do
                if part:IsA("BasePart") then
                    if xrayApplied[part] == nil then
                        xrayApplied[part] = part.LocalTransparencyModifier
                    end
                    part.LocalTransparencyModifier = Flags["X-Ray Trans"] or 0.65
                end
            end
        end
    end
    getgenv().IMP_xrayUpdate = xrayUpdate

    local worldAcc = 0
    getgenv().IMP_worldUpdate = function(dt)
        if not Flags["Fullbright"] and not Flags["Time Changer"] and not Flags["Custom Atmos"] then return end
        worldAcc = worldAcc + (dt or 0.016)
        if worldAcc < 0.25 then return end
        worldAcc = 0
        if Flags["Custom Atmos"] then applySimpleAtmos() end
        if Flags["Fullbright"] then
            saveLighting()
            fbCC.Enabled = true
            fbCC.Brightness = Flags["Brightness"] or 0.1
            Lighting.Ambient = Color3.new(1, 1, 1)
            Lighting.OutdoorAmbient = Color3.new(1, 1, 1)
            Lighting.GlobalShadows = false
            Lighting.ExposureCompensation = (Flags["Brightness"] or 0.1) * 2
        end
        forceTime()
    end
end

-- ========================================== UNLOAD ==========================================
local function doUnload()
    pcall(getgenv().IMP_stopAutoWalk)
    pcall(getgenv().IMP_applyIR)
    pcall(getgenv().IMP_applyFastImmediate)
    pcall(getgenv().IMP_RestoreGunChams)
    pcall(getgenv().IMP_setNoFog, false)
    pcall(getgenv().IMP_restoreLighting)
    pcall(function()
        local ao = getgenv().IMP_atmosOriginal
        Lighting.Ambient = ao.Ambient
        Lighting.OutdoorAmbient = ao.OutdoorAmbient
        Lighting.FogColor = ao.FogColor
        Lighting.FogStart = ao.FogStart
        Lighting.FogEnd = ao.FogEnd
    end)
    pcall(function() getgenv().IMP_setVeg("Grass", false) end)
    pcall(function() getgenv().IMP_setVeg("Leaves", false) end)
    if Camera then pcall(function() Camera.FieldOfView = 70 end) end
    pcall(function() getgenv().IMP_silFovCircle:Remove() end)
    pcall(function() getgenv().IMP_silTargetLine:Remove() end)
    local chLines = getgenv().IMP_chLines
    if chLines then
        for i = 1, 4 do pcall(function() chLines[i]:Remove() end) end
    end
    if getgenv().Imperium_CrosshairTarget_Cleanup then pcall(getgenv().Imperium_CrosshairTarget_Cleanup) end
    if getgenv().IMP_hoverInvCleanup then pcall(getgenv().IMP_hoverInvCleanup) end
    pcall(function() getgenv().IMP_ScreenGui:Destroy() end)
end
getgenv().IMP_unload = doUnload

-- ========================================== MAIN LOOP ==========================================
do
    local irAcc = 0
    local accXray, accWorld, accMove, accGunChams = 0, 0, 0, 0
    local THROTTLE = 0.033

    RunService.RenderStepped:Connect(function(dt)
        if not getgenv().IMP_Flags then return end
        pcall(getgenv().IMP_fovApplyNow)
        pcall(getgenv().IMP_updateCrosshair)
        if getgenv().IMP_targetHudUpdate then pcall(getgenv().IMP_targetHudUpdate) end
        if getgenv().IMP_espUpdate then pcall(getgenv().IMP_espUpdate) end
        pcall(getgenv().IMP_hoverInvUpdate, dt)
        pcall(getgenv().IMP_freecamUpdate, dt)
        pcall(getgenv().IMP_triggerbotUpdate)
        pcall(getgenv().IMP_silentUpdate)
    end)

    RunService.Heartbeat:Connect(function(dt)
        if not getgenv().IMP_Flags then return end
        accXray = accXray + dt
        if accXray >= THROTTLE then
            accXray = 0
            if getgenv().IMP_xrayUpdate then pcall(getgenv().IMP_xrayUpdate, dt) end
        end
        accWorld = accWorld + dt
        if accWorld >= THROTTLE then
            accWorld = 0
            if getgenv().IMP_worldUpdate then pcall(getgenv().IMP_worldUpdate, dt) end
        end
        accMove = accMove + dt
        if accMove >= THROTTLE then
            accMove = 0
            if getgenv().IMP_moveUpdate then pcall(getgenv().IMP_moveUpdate, dt) end
        end
        accGunChams = accGunChams + dt
        if accGunChams >= THROTTLE then
            accGunChams = 0
            if getgenv().IMP_GunChamsUpdate then pcall(getgenv().IMP_GunChamsUpdate, dt) end
        end
        if getgenv().IMP_Flags["Instant Reload"] then
            irAcc = irAcc + dt
            if irAcc >= 2 then
                irAcc = 0
                pcall(getgenv().IMP_applyIR)
            end
        end
    end)
end
